// src/main/proxyChainConfigGenerator.ts
public generateChainConfig(nodes: ProxyNode[]): ProxyChainConfig {
  // 为每个节点分配端口
  const chainNodes: ChainNodeConfig[] = [];
  for (let i = 0; i < nodes.length; i++) {
    const node = nodes[i];
    if (!node) continue;
    
    const localPort = this.allocatePort();
    const upstreamPort = i > 0 && chainNodes[i - 1] ? chainNodes[i - 1]!.localPort : undefined;
    
    chainNodes.push({ node, localPort, upstreamPort });
  }
  
  // 生成入站、出站和路由配置
  const inbounds = chainNodes.length > 0 && chainNodes[0] ? this.generateInbounds(chainNodes[0]) : [];
  const outbounds = this.generateOutbounds(chainNodes);
  const route = this.generateRoute(chainNodes);
  
  return { inbounds, outbounds, route, log: this.generateLogConfig() };
}
```

**ProxyManager中的链式启动方法**

// src/main/proxyManager.ts
public async startChain(chainConfig: ChainConfig, networkSettings?: NetworkSettings): Promise<void> {
  let nodes: ProxyNode[] = [];
  
  if (chainConfig.type === 'static') {
    nodes = await this.getNodesByIds(chainConfig.proxies);
  } else if (chainConfig.type === 'dynamic') {
    nodes = await this.getNodesFromSubscriptions(chainConfig.proxies);
  }
  
  // 生成代理链配置
  const chainProxyConfig = proxyChainConfigGenerator.generateChainConfig(nodes);
  
  // 应用网络设置并启动
  const finalConfig = this.applyNetworkSettingsToConfig(chainProxyConfig, networkSettings);
  await this.startSingbox(finalConfig, networkSettings);
}
```

## 6. 技术架构详解 (Technical Architecture Details)

### 代理链工作原理
```
用户流量 → 本地端口1080 → 节点A → 节点B → 节点C → 目标网站
```

### 核心组件
1. **ProxyChainConfigGenerator**: 负责生成真正的链式配置
2. **端口管理器**: 为每个中间节点分配独立端口
3. **配置生成器**: 生成符合sing-box规范的配置文件
4. **动态选择器**: 从订阅分组中选择最佳节点

### 配置结构
- **入站配置**: 监听本地端口，接收用户流量
- **出站配置**: 每个节点连接到下一个节点或目标
- **路由配置**: 将流量路由到代理链的第一个节点
- **日志配置**: 记录代理链运行状态
