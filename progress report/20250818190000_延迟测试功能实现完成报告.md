```typescript
public async testNodeLatencyViaMainProcess(node: ProxyServer, config?: Partial<LatencyTestConfig>): Promise<LatencyTestResult> {
  const testConfig = { ...this.defaultConfig, ...config };
  
  try {
    const result = await window.electron.ipcRenderer.invoke('proxy:testLatency', {
      node,
      config: testConfig
    });
    
    if (result.success) {
      return {
        latency: result.latency,
        success: true,
        timestamp: Date.now()
      };
    } else {
      throw new Error(result.error || 'Unknown error');
    }
  } catch (error) {
    return {
      latency: 0,
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error',
      timestamp: Date.now()
    };
  }
}
```

**主进程延迟测试处理**:
```typescript
ipcMain.handle('proxy:testLatency', async (_, { node, config }) => {
  try {
    const startTime = Date.now();
    const testUrl = config.testUrl || 'http://connectivitycheck.gstatic.com/generate_204';
    const timeout = config.timeout || 10000;
    
    return new Promise((resolve) => {
      const url = new URL(testUrl);
      const isHttps = url.protocol === 'https:';
      const client = isHttps ? https : http;
      
      const req = client.request(url, {
        method: 'GET',
        timeout: timeout,
      }, (res: any) => {
        const endTime = Date.now();
        const latency = endTime - startTime;
        
        resolve({
          success: true,
          latency,
          statusCode: res.statusCode
        });
      });
      
      req.on('error', (error: any) => {
        resolve({
          success: false,
          error: error.message,
          latency: 0
        });
      });
      
      req.end();
    });
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error',
      latency: 0
    };
  }
});
```
