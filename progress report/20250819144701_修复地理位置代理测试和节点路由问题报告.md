```typescript
// src/main/index.ts - 修复前的问题代码
const client = isHttps ? https : http;  // 错误：按目标URL协议选择客户端
const proxyReq = client.request(proxyOptions, (proxyRes) => {
  // 用 https 去连 HTTP 代理会握手失败，导致 ECONNRESET
});

// 修复后的正确实现
const connectReq = http.request({  // 始终用 HTTP 连接代理
  host: proxyHost,
  port: proxyPort,
  method: 'CONNECT',
  path: `${targetUrl.hostname}:${targetPort}`,
  headers: { Host: `${targetUrl.hostname}:${targetPort}` }
});

connectReq.on('connect', (_res, socket) => {
  if (targetIsHttps) {
    const tlsSocket = tls.connect({ socket, servername: targetUrl.hostname });
    tlsSocket.once('secureConnect', () => {  // 等待TLS握手完成后再发送请求
      tlsSocket.write(requestLines);
    });
  } else {
    socket.write(requestLines);  // 明文HTTP直接在socket上发送
  }
});
```
