```typescript
// src/main/dynamicChainManager.ts
export class DynamicChainManager {
  private static instance: DynamicChainManager;
  private activeTimers: Map<string, NodeJS.Timeout> = new Map();

  public async startChain(chainConfig: ChainConfig): Promise<void> {
    if (chainConfig.type !== 'dynamic' || !chainConfig.proxies) {
      throw new Error('Invalid dynamic chain configuration.');
    }

    console.log(`[DynamicChainManager] Starting dynamic chain: ${chainConfig.name}`);
    await this.updateAndApplyChain(chainConfig);

    // 设置定时更新
    const updateInterval = 300000; // 5分钟
    const timer = setInterval(() => {
      console.log(`[DynamicChainManager] Performing scheduled update for chain: ${chainConfig.name}`);
      this.updateAndApplyChain(chainConfig);
    }, updateInterval);

    this.activeTimers.set(chainConfig.id, timer);
  }
}
```

**示例: 节点分组逻辑修复**
```typescript
// src/renderer/pages/ProxyManagement.tsx 和 src/renderer/components/NodeSelector.tsx
const loadAndGroupNodes = () => {
  const subscriptions = Storage.get<Subscription[]>(STORAGE_KEYS.SUBSCRIPTION_CONFIG, []) || [];
  
  // 从订阅数据中生成节点数据，并设置正确的 subscriptionId
  const nodesFromSubscriptions: ProxyNode[] = [];
  subscriptions.forEach(subscription => {
    if (subscription.enabled && subscription.servers) {
      subscription.servers.forEach(server => {
        if (server.enabled) {
          const node: ProxyNode = {
            id: server.id,
            name: server.name,
            type: server.protocol as any,
            server: server.host,
            port: server.port,
            subscriptionId: subscription.id, // 设置订阅ID
            // ... 其他属性
          };
          nodesFromSubscriptions.push(node);
        }
      });
    }
  });

  // 合并从订阅生成的节点和现有的手动添加的节点
  const manualNodes = nodes.filter(node => !node.subscriptionId);
  const allNodes = [...nodesFromSubscriptions, ...manualNodes];

  // 按订阅ID分组
  const grouped = _.groupBy(allNodes, (node: ProxyNode) => node.subscriptionId || 'ungrouped');
  setGroupedNodes(grouped);
};
```

## 7. 重要修复记录 (Critical Fixes)

### 节点分组显示问题修复 (2025-08-21)
**问题描述**: 用户反馈所有节点都显示在"未分组"类别下，无法正确按订阅分组显示。

**根本原因**: 
1. `ProxyNode` 接口中缺少 `subscriptionId` 字段
2. 节点数据与订阅数据分离存储，但缺少正确的关联逻辑
3. `NodeSelector` 和 `ProxyManagement` 组件中的分组逻辑都存在问题

**修复方案**:
1. 在 `ProxyNode` 接口中添加 `subscriptionId?: string` 字段
2. 修改 `loadAndGroupNodes` 函数，从订阅数据中动态生成节点数据并设置正确的 `subscriptionId`
3. 同步修复 `NodeSelector` 组件中的分组逻辑
4. 确保手动添加的节点和订阅节点能够正确合并和分组

**修复效果**: ✅ 节点现在能够正确按订阅分组显示，用户可以看到类似"STC (41个节点)"和"手动添加/未分组 (142个节点)"的分组

### 动态代理链保存问题修复 (2025-08-21)
**问题描述**: 用户反馈无法保存动态代理链，"保存"按钮被禁用，且代理链计数显示不正确。

**根本原因**: 
1. 保存按钮的 `disabled` 条件只检查了静态链的 `chainNodes.length`，没有考虑动态链的 `dynamicChainProxies.length`
2. 代理链计数显示只显示了 `chainNodes.length`，没有根据链类型显示正确的计数
3. 清空按钮的逻辑没有处理动态链的 `dynamicChainProxies` 状态

**修复方案**:
1. 修改保存按钮的 `disabled` 条件，同时检查链名称、静态链节点数量和动态链订阅分组数量
2. 修复代理链计数显示，根据当前链类型显示正确的数量
3. 修复清空按钮逻辑，确保同时清空静态链和动态链的状态
4. 修复清空按钮的禁用条件，使其能正确处理两种链类型

**修复效果**: ✅ 现在用户可以正常保存动态代理链，按钮状态和计数显示都正确反映当前链的内容
