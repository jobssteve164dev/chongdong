## 2. 核心实现 (Core Implementation)

### a. 方法论/设计思路
采用了分层架构设计，将协议引擎管理、系统代理控制和链式代理配置完全分离。前端通过IPC与主进程通信，主进程负责管理外部代理引擎进程和系统级网络设置。采用了Sing-box + Xray双引擎架构，以Sing-box作为主要引擎处理链式代理，Xray作为辅助引擎处理特殊协议。

```typescript
// src/renderer/utils/proxyEngine.ts
export class ProxyEngine {
  public async start(config: ProxyConfig): Promise<void> {
    switch (config.type) {
      case 'singbox':
        await this.startSingbox(config);
        break;
      case 'xray':
        await this.startXray(config);
        break;
      case 'clash':
        await this.startClash(config);
        break;
    }
  }
}
```

```typescript
public async setSystemProxy(host: string, port: number): Promise<void> {
  switch (process.platform) {
    case 'win32':
      await this.setWindowsProxy(host, port);
      break;
    case 'darwin':
      await this.setMacOSProxy(host, port);
      break;
    case 'linux':
      await this.setLinuxProxy(host, port);
      break;
  }
}
```

```typescript
// src/renderer/utils/chainProxy.ts
public generateSingboxConfig(chain: ChainConfig, proxyConfigs: ProxyConfig[]): any {
  const outbounds: any[] = [];
  // 创建代理出站
  for (const proxyId of chain.proxies) {
    const proxy = proxyMap.get(proxyId);
    if (proxy) {
      const outbound = this.convertProxyToSingboxOutbound(proxy);
      outbounds.push(outbound);
    }
  }
  // 创建链式出站
  if (outbounds.length > 1) {
    const chainOutbound = {
      type: "chain",
      tag: `chain-${chain.id}`,
      outbounds: outbounds.map(o => o.tag)
    };
    outbounds.push(chainOutbound);
  }
  return { outbounds, route: { rules, final: `chain-${chain.id}` } };
}
```
