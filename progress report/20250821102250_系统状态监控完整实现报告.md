```typescript
export class SystemMonitor {
  /**
   * 获取CPU使用率
   */
  private async getCpuUsage(): Promise<number> {
    const cpus = os.cpus();
    // 计算CPU使用率的逻辑
    const usage = 100 - (100 * idle / total);
    return Math.round(usage * 100) / 100;
  }

  /**
   * 获取内存使用率
   */
  private getMemoryUsage(): number {
    const totalMem = os.totalmem();
    const freeMem = os.freemem();
    const usage = ((totalMem - freeMem) / totalMem) * 100;
    return Math.round(usage * 100) / 100;
  }

  /**
   * 获取系统代理状态
   */
  private async getSystemProxyStatus(): Promise<{ enabled: boolean; host?: string; port?: number }> {
    // 跨平台系统代理检测逻辑
    switch (process.platform) {
      case 'darwin': return await this.getMacOSProxyStatus();
      case 'win32': return await this.getWindowsProxyStatus();
      case 'linux': return await this.getLinuxProxyStatus();
    }
  }
}
```

## 5. 自我评估与学习 (Self-Assessment & Learning)

*   **遇到的挑战**: 
    - TypeScript严格类型检查导致可选属性赋值错误，需要特殊处理undefined值
    - Electron渲染进程中不能直接使用Node.js API，需要通过预加载脚本暴露的IPC接口

*   **学到的教训**: 
    - 在Electron应用中，主进程和渲染进程的职责分离很重要，系统级操作应在主进程中执行
    - TypeScript的exactOptionalPropertyTypes配置要求更严格的类型处理，需要避免将undefined赋值给可选属性
    - 跨平台系统监控需要针对不同操作系统实现不同的检测逻辑
