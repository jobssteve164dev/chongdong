### c. 关键代码片段

**示例: 统一泄露防护管理器**
```typescript
// src/main/services/leakProtectionManager.ts
export class LeakProtectionManager {
  public async checkAllLeaks(): Promise<{
    dns: DnsLeakResult;
    ipv6: Ipv6LeakResult;
    webRTC: WebRTCLeakResult;
  }> {
    const [dnsResult, ipv6Result, webRTCResult] = await Promise.all([
      dnsService.checkDnsLeak(),
      ipv6LeakProtectionService.checkIPv6Leak(),
      webRTCLeakProtectionService.checkWebRTCLeak()
    ]);
    return { dns: dnsResult, ipv6: ipv6Result, webRTC: webRTCResult };
  }
}
```
