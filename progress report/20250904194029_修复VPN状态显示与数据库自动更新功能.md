```typescript
// src/main/tunController.ts
static async isRunning(): Promise<boolean> {
  try {
    // 原有PID检查逻辑...
    
    // 新增备用检测方式
    if (process.platform === 'darwin') {
      // macOS: 检查utun设备
      const { exec } = require('child_process');
      const { promisify } = require('util');
      const execAsync = promisify(exec);
      
      try {
        const { stdout } = await execAsync('ifconfig utun*');
        return stdout.includes('utun') && stdout.includes('UP');
      } catch (error) {
        console.warn('ifconfig检查失败:', error);
      }
    }
    
    // 通用进程检查
    try {
      const { exec } = require('child_process');
      const { promisify } = require('util');
      const execAsync = promisify(exec);
      
      const command = process.platform === 'win32' 
        ? 'tasklist /FI "IMAGENAME eq tun2socks.exe"' 
        : 'pgrep tun2socks';
      
      const { stdout } = await execAsync(command);
      return stdout.trim().length > 0;
    } catch (error) {
      console.warn('进程检查失败:', error);
    }
    
    return false;
  } catch (error) {
    console.error('TUN状态检查失败:', error);
    return false;
  }
}
```

**数据库自动更新管理器**
```typescript
// src/main/databaseUpdateManager.ts
export class DatabaseUpdateManager {
  private static instance: DatabaseUpdateManager;
  private updateTimer: NodeJS.Timeout | null = null;
  
  static getInstance(): DatabaseUpdateManager {
    if (!DatabaseUpdateManager.instance) {
      DatabaseUpdateManager.instance = new DatabaseUpdateManager();
    }
    return DatabaseUpdateManager.instance;
  }
  
  startAutoUpdate(settings: any): void {
    this.stopAutoUpdate();
    
    if (settings.enableDatabaseAutoUpdate) {
      const interval = (settings.databaseUpdateInterval || 24) * 60 * 60 * 1000; // 转换为毫秒
      this.updateTimer = setInterval(() => {
        this.checkForUpdates(settings);
      }, interval);
      
      // 启动时检查
      if (settings.databaseUpdateCheckOnStartup) {
        setTimeout(() => this.checkForUpdates(settings), 5000);
      }
    }
  }
  
  async checkForUpdates(settings: any): Promise<UpdateCheckResult> {
    const results: UpdateCheckResult = { success: true, updates: [] };
    
    for (const db of [geoipDatabase, geositeDatabase]) {
      const updateResult = await this.checkDatabaseUpdate(db);
      if (updateResult.hasUpdate) {
        results.updates.push(updateResult);
        await this.downloadDatabase(db);
      }
    }
    
    return results;
  }
}
```
