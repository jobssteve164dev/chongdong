```typescript
// 支持多种订阅格式
private detectFormat(content: string): string {
  if (this.isBase64(content)) return 'base64';
  if (content.includes('proxies:')) return 'clash';
  if (content.includes('"outbounds"')) return 'singbox';
  if (content.includes('"protocol"')) return 'v2ray';
  if (content.includes('vmess://')) return 'links';
  return 'unknown';
}
```
